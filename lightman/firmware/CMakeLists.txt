# @file CMakeLists.txt
# @brief Build the Pico firmware or portable host tests and generate embedded font tables.
cmake_minimum_required(VERSION 3.24)
option(LIGHTMAN_HOST_TESTS "Build portable tests without the Pico SDK" OFF)
if(NOT LIGHTMAN_HOST_TESTS)
    if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
        message(FATAL_ERROR "Set PICO_SDK_PATH to an installed Pico SDK (tested with 2.2.0)")
    endif()
    set(PICO_BOARD pico CACHE STRING "Original RP2040 Pico")
    include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
endif()
project(lightman VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(lightman_core src/terminal.cpp src/keyboard.cpp src/config.cpp)
target_include_directories(lightman_core PUBLIC include)
target_compile_options(lightman_core PRIVATE -Wall -Wextra -Wpedantic)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(LIGHTMAN_REGULAR_FONT "${CMAKE_CURRENT_SOURCE_DIR}/../assets/fonts/lightman-regular.psf" CACHE FILEPATH "Regular Unicode 8x16 PSF font")
set(LIGHTMAN_BOLD_FONT "${CMAKE_CURRENT_SOURCE_DIR}/../assets/fonts/lightman-bold.psf" CACHE FILEPATH "Bold Unicode 8x16 PSF font")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fonts.h"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/psf_to_header.py"
        "${LIGHTMAN_REGULAR_FONT}" "${LIGHTMAN_BOLD_FONT}" "${CMAKE_CURRENT_BINARY_DIR}/fonts.h"
    DEPENDS ../tools/psf_to_header.py "${LIGHTMAN_REGULAR_FONT}" "${LIGHTMAN_BOLD_FONT}" VERBATIM)
add_custom_target(lightman_fonts DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/fonts.h")
if(LIGHTMAN_HOST_TESTS)
    enable_testing()
    add_executable(lightman_tests ../tests/core_tests.cpp)
    target_link_libraries(lightman_tests PRIVATE lightman_core)
    target_compile_options(lightman_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    add_dependencies(lightman_tests lightman_fonts)
    target_include_directories(lightman_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME terminal_keyboard_settings COMMAND lightman_tests)
else()
    pico_sdk_init()
    if(NOT PICO_BOARD STREQUAL "pico")
        message(FATAL_ERROR "This pin map and settings layout target the original 2 MiB Pico")
    endif()
    add_executable(lightman src/main.cpp src/video.cpp src/serial.cpp src/ps2.cpp src/settings.cpp)
    add_dependencies(lightman lightman_fonts)
    target_include_directories(lightman PRIVATE boards "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(lightman PRIVATE -O2 -Wall -Wextra $<$<COMPILE_LANGUAGE:CXX>:-Wpedantic>)
    target_compile_definitions(lightman PRIVATE PICO_STACK_SIZE=0x1000 PICO_CORE1_STACK_SIZE=0x1000)
    target_compile_options(lightman_core PRIVATE -O2)
    target_link_libraries(lightman PRIVATE lightman_core pico_stdlib pico_multicore hardware_pio hardware_dma hardware_uart hardware_flash hardware_sync)
    target_link_options(lightman PRIVATE "LINKER:-T,${CMAKE_CURRENT_SOURCE_DIR}/flash_limit.ld")
    pico_generate_pio_header(lightman "${CMAKE_CURRENT_SOURCE_DIR}/src/video.pio")
    pico_enable_stdio_uart(lightman 0)
    pico_enable_stdio_usb(lightman 0)
    pico_set_program_name(lightman "Lightman serial terminal")
    pico_set_program_version(lightman "${PROJECT_VERSION}")
    pico_add_extra_outputs(lightman)
endif()
